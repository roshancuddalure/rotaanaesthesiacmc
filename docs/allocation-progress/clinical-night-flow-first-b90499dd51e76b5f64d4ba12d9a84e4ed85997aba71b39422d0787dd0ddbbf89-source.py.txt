"""Bounded read-only prototype: retained flow witness plus canonical checks.

Guided cuts remove individual nonoriginal edges, so exhausting this heuristic is
not an infeasibility proof. Accepted state is hypothetical, never an Apply token.
"""
from collections import Counter, defaultdict
from time import perf_counter

from app.services import balance_duties as bd
from app.services.allocation_fairness_policy import fairness_mode
from app.services.rota_optimizer import incompatible
from study_night_count_cost import analyze


def snapshot(state):
    return {aid: (row['person'], state.slot_unit[row['slot'].id]) for aid, row in state.rows.items()}


def measures(state, people, holders=None, aids=None):
    values = {p: Counter() for p in people}
    chosen = {aid: holders[i] for i, aid in enumerate(aids)} if holders is not None else {}
    for aid, row in state.rows.items():
        p = chosen.get(aid, row['person'])
        if p not in values:
            continue
        slot = row['slot']
        values[p].update(duties=1, minutes=round((slot.ends_at-slot.starts_at).total_seconds()/60),
                         night=int(slot.is_24hr or slot.starts_at.hour >= 18 or slot.ends_at.date() > slot.starts_at.date()),
                         weekend=int(slot.duty_date.weekday() >= 5))
    return {key: dict(gap=max(v[key] for v in values.values())-min(v[key] for v in values.values()),
                      squares=sum(v[key]**2 for v in values.values()))
            for key in ('duties', 'minutes', 'night', 'weekend')}


def temporal_cut(state, aids, rows, holders):
    by_person = defaultdict(list)
    # Earlier draft moves use provisional entries with assignment_id=None.
    # The unique person/slot identity still identifies each replaced baseline row.
    selected = {(state.rows[aid]['slot'].id, state.rows[aid]['person']) for aid in aids}
    for p, entries in state.ctx.entries_by_person.items():
        for entry in entries:
            if (entry.slot_id, p) not in selected:
                by_person[p].append((None, entry))
    for i, aid in enumerate(aids):
        by_person[holders[i]].append((i, state.rows[aid]['slot']))
    for p in sorted(by_person, key=str):
        duties = by_person[p]
        for offset, (i, first) in enumerate(duties):
            for j, second in duties[offset+1:]:
                changed = [k for k in (i, j) if k is not None and holders[k] != rows[k]['original']]
                if changed and incompatible(first, second, state.rules):
                    return max(changed), p
    return None


def audit_holders(state):
    for aid, row in list(state.rows.items()):
        person, unit = row['person'], state.slot_unit[row['slot'].id]
        state.remove(aid)
        try:
            actual_unit, code = state.place_check(person, row['slot'], allow_unposted=bool(row.get('extra')))
        finally:
            state.add(aid, person, unit)
        if actual_unit != unit or code is not None:
            return code or 'unit_mismatch'
    return None


def study(state, *, level='3RD_CALL', max_trials=64, max_seconds=60, max_changed=64):
    assert fairness_mode(state.rules) == 'equal_counts'
    people = set(state.members_by_level.get(level, []))
    assert people and len({state.share[p] for p in people}) == 1
    original, credits = snapshot(state), dict(state.extra_count)
    immutable = {aid for aid, row in state.rows.items() if row['person'] not in people
                 or not row['movable'] or row.get('extra')}
    before = measures(state, people)
    counts = {p: state.duties.get(p, 0) for p in people}
    low, high = min(counts.values()), max(counts.values())
    assert high-low <= 2
    bounds = {p: (max(low, state.goal[p]-state.gap, 0), min(high, state.goal[p]+state.gap)) for p in people}
    aids, rows = [], []
    for aid, row in state.rows.items():
        if row['person'] not in people:
            continue
        fixed = not row['movable'] or bool(row.get('extra'))
        aids.append(aid)
        rows.append(dict(original=row['person'], fixed=fixed,
            allowed=[row['person']] if fixed else sorted((p for p in people
                if state.structural_block(p, row['slot']) is None), key=str),
            night=bool(row['slot'].is_24hr or row['slot'].starts_at.hour >= 18
                       or row['slot'].ends_at.date() > row['slot'].starts_at.date())))
    started = perf_counter()
    refused, attempts = Counter(), []
    accepted = False
    accepted_rows = 0
    stop = 'trial_limit'

    def ban(i, person, code):
        assert person != rows[i]['original'] and not rows[i]['fixed']
        rows[i]['allowed'].remove(person)
        refused[code] += 1

    def restore(planner):
        current = snapshot(state)
        planner._raw([(aid, person, unit) for aid, (person, unit) in original.items()
                      if current[aid] != (person, unit)])
        state.invalidate_safety(state.slots)
        assert snapshot(state) == original

    for trial in range(max_trials):
        remaining = max_seconds-(perf_counter()-started)
        if remaining <= 0:
            stop = 'observation_budget'
            break
        holders = {}
        result = analyze(rows, counts, count_bounds=bounds, gap=2, seconds=min(2, remaining),
                         prefer_original=True, witness=holders)
        observation = dict(trial=trial+1, count_cost=result['minimum_squared_counts'],
                           changes=result['changed_assignments'], structural_status=result['status'])
        attempts.append(observation)
        if not holders:
            stop = 'no_more_structural_witnesses_in_cut_model'
            break
        if result['minimum_squared_counts'] > before['duties']['squares']:
            stop = 'higher_count_cost_in_cut_model'
            break
        changed = [i for i in holders if holders[i] != rows[i]['original']]
        if not changed or len(changed) > max_changed:
            stop = 'changed_row_bound'
            break
        cut = temporal_cut(state, aids, rows, holders)
        if cut:
            ban(*cut, 'temporal_guided_cut')
            observation['refusal'] = 'temporal_guided_cut'
            continue
        proposed = measures(state, people, holders, aids)
        if any(proposed[key][measure] > before[key][measure]
               for key in ('minutes', 'weekend') for measure in ('gap', 'squares')):
            # This is a heuristic cut, never proof that a safe witness is absent.
            person = max((holders[i] for i in changed), key=lambda p: (
                sum(1 for i in changed if holders[i] == p), str(p)))
            i = next(i for i in reversed(changed) if holders[i] == person)
            ban(i, person, 'burden_guided_cut')
            observation['refusal'] = 'burden_guided_cut'
            continue
        planner = bd._Planner(state, bd.MAX_WORK)
        pairs = sorted(((aids[i], holders[i]) for i in changed),
                       key=lambda pair: (state.rows[pair[0]]['slot'].starts_at, str(pair[0])))
        ok, code, placed = planner.transfer(pairs)
        if ok:
            try:
                current = snapshot(state)
                code = ('protected_or_outside_level_changed' if any(current[a] != original[a] for a in immutable)
                        else audit_holders(state))
            except Exception:
                restore(planner)
                raise
            if code:
                restore(planner)
                ok = False
        if not ok:
            assert snapshot(state) == original and dict(state.extra_count) == credits
            i = changed[-1]
            ban(i, holders[i], code or 'atomic_placement_refusal')
            observation['refusal'] = code or 'atomic_placement_refusal'
            continue
        assert len(placed) == len(changed) and dict(state.extra_count) == credits
        after = measures(state, people)
        assert after['duties']['gap'] <= before['duties']['gap'] and after['duties']['squares'] <= before['duties']['squares']
        assert after['night']['gap'] <= 2
        assert all(after[key][metric] <= before[key][metric]
                   for key in ('minutes', 'weekend') for metric in ('gap', 'squares'))
        accepted, accepted_rows, stop = True, len(placed), 'strict_placement_and_all_holder_replay_passed'
        break
    if not accepted:
        assert snapshot(state) == original and dict(state.extra_count) == credits
    return dict(status='offline_clinical_prototype_only', accepted=accepted,
                changed_rows=accepted_rows, call_level=level, before=before, after=measures(state, people),
                trials=len(attempts), trial_limit=max_trials, elapsed_seconds=perf_counter()-started,
                observation_limit_seconds=max_seconds, changed_row_limit=max_changed,
                stop=stop, refusals=dict(refused), attempts=attempts, operational_writes=0,
                final_independent_validation_required=True,
                limitations='Heuristic individual-edge cuts, not a global search or infeasibility proof. No Apply token; all840 independent checks still required before any promotion.')
